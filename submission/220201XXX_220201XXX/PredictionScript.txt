import os
from pathlib import Path

import torch
from PIL import Image, ImageOps, UnidentifiedImageError
from torchvision import transforms
from torchvision.models import efficientnet_b0


MODEL_FILENAME = "best_efficientnet_b0.pt"
IMAGE_SUFFIXES = {".jpg", ".jpeg", ".png", ".webp", ".bmp"}
IMAGENET_MEAN = [0.485, 0.456, 0.406]
IMAGENET_STD = [0.229, 0.224, 0.225]
DEFAULT_CLASS_NAMES = [
    "F1",
    "HATCHBACK",
    "MICRO",
    "PICK_UP",
    "SEDAN",
    "STATION_WAGON",
    "SUV",
    "VAN",
]
CLASS_NAME_TO_OFFICIAL_LABEL = {
    "SUV": 1,
    "VAN": 2,
    "STATION_WAGON": 3,
    "MICRO": 4,
    "F1": 5,
    "SEDAN": 6,
    "HATCHBACK": 7,
    "PICK_UP": 8,
}


def _base_dir():
    if "__file__" in globals():
        return Path(__file__).resolve().parent
    return Path.cwd()


def _find_model_path():
    candidates = [
        _base_dir() / MODEL_FILENAME,
        Path.cwd() / MODEL_FILENAME,
        Path("/content") / MODEL_FILENAME,
    ]
    for candidate in candidates:
        if candidate.exists():
            return candidate

    for root in [Path.cwd(), Path("/content")]:
        if root.exists():
            matches = list(root.rglob(MODEL_FILENAME))
            if matches:
                return matches[0]

    raise FileNotFoundError(
        f"Could not find {MODEL_FILENAME}. Put it in the same folder as PredictionScript.txt."
    )


def _load_checkpoint(path, device):
    try:
        return torch.load(path, map_location=device, weights_only=False)
    except TypeError:
        return torch.load(path, map_location=device)


def _build_model(num_classes, dropout):
    model = efficientnet_b0(weights=None)
    in_features = model.classifier[-1].in_features
    model.classifier = torch.nn.Sequential(
        torch.nn.Dropout(p=dropout, inplace=False),
        torch.nn.Linear(in_features, num_classes),
    )
    return model


def _letterbox_image(image, size, background="white"):
    image = ImageOps.exif_transpose(image)
    has_alpha = image.mode in {"RGBA", "LA"} or (image.mode == "P" and "transparency" in image.info)

    if has_alpha:
        rgba_image = image.convert("RGBA")
        bg_color = (255, 255, 255, 255) if background == "white" else (0, 0, 0, 255)
        canvas = Image.new("RGBA", rgba_image.size, bg_color)
        image = Image.alpha_composite(canvas, rgba_image).convert("RGB")
    else:
        image = image.convert("RGB")

    image.thumbnail((size, size), Image.Resampling.LANCZOS)
    canvas = Image.new("RGB", (size, size), background)
    x = (size - image.width) // 2
    y = (size - image.height) // 2
    canvas.paste(image, (x, y))
    return canvas


def _make_preprocess(image_size):
    return transforms.Compose(
        [
            transforms.Lambda(lambda image: _letterbox_image(image, image_size, "white")),
            transforms.ToTensor(),
            transforms.Normalize(IMAGENET_MEAN, IMAGENET_STD),
        ]
    )


def _image_files(test_root):
    return sorted(
        [path for path in test_root.rglob("*") if path.is_file() and path.suffix.lower() in IMAGE_SUFFIXES],
        key=lambda path: str(path.relative_to(test_root)).lower(),
    )


def _load_model():
    device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
    checkpoint_path = _find_model_path()
    checkpoint = _load_checkpoint(checkpoint_path, device)

    checkpoint_args = checkpoint.get("args", {})
    class_names = checkpoint.get("class_names", DEFAULT_CLASS_NAMES)
    image_size = int(checkpoint.get("image_size", checkpoint_args.get("image_size", 320)))
    dropout = float(checkpoint_args.get("dropout", 0.35))

    model = _build_model(len(class_names), dropout)
    model.load_state_dict(checkpoint["model_state_dict"])
    model.to(device)
    model.eval()
    return model, class_names, image_size, device


def _write_predictions(image_names, preds):
    lines = [f"{image_names[i]} | Pred: {preds[i]}" for i in range(len(preds))]

    output_paths = [Path.cwd() / "Preds.txt", Path.cwd() / "preds.txt"]
    base_output = _base_dir() / "Preds.txt"
    if base_output not in output_paths:
        output_paths.append(base_output)

    unique_paths = []
    seen = set()
    for output_path in output_paths:
        key = str(output_path).lower() if os.name == "nt" else str(output_path)
        if key in seen:
            continue
        seen.add(key)
        unique_paths.append(output_path)

    for output_path in unique_paths:
        output_path.write_text("\n".join(lines) + ("\n" if lines else ""), encoding="utf-8")

    for line in lines:
        print(line)

    print(f"Saved {len(lines)} predictions to Preds.txt")


def Predict(file_path):
    test_root = Path(file_path)
    if not test_root.exists():
        raise FileNotFoundError(f"Test data folder not found: {test_root}")

    if not any(path.is_file() and path.suffix.lower() in IMAGE_SUFFIXES for path in test_root.rglob("*")):
        child_dirs = [path for path in test_root.iterdir() if path.is_dir()]
        if len(child_dirs) == 1:
            nested_root = child_dirs[0]
            if any(path.is_file() and path.suffix.lower() in IMAGE_SUFFIXES for path in nested_root.rglob("*")):
                test_root = nested_root

    model, class_names, image_size, device = _load_model()
    preprocess = _make_preprocess(image_size)
    image_paths = _image_files(test_root)
    if not image_paths:
        raise RuntimeError(f"No readable images found under: {test_root}")

    image_names = []
    preds = []
    batch = []
    batch_names = []
    batch_size = 64

    with torch.no_grad():
        for image_path in image_paths:
            try:
                with Image.open(image_path) as image:
                    batch.append(preprocess(image).unsqueeze(0))
                    batch_names.append(image_path.name)
            except (OSError, UnidentifiedImageError, ValueError) as exc:
                print(f"Skipping unreadable image: {image_path} ({exc})")
                continue

            if len(batch) >= batch_size:
                inputs = torch.cat(batch, dim=0).to(device)
                logits = model(inputs)
                pred_indices = logits.argmax(dim=1).detach().cpu().tolist()
                for name, pred_index in zip(batch_names, pred_indices):
                    class_name = class_names[pred_index]
                    image_names.append(name)
                    preds.append(CLASS_NAME_TO_OFFICIAL_LABEL[class_name])
                batch = []
                batch_names = []

        if batch:
            inputs = torch.cat(batch, dim=0).to(device)
            logits = model(inputs)
            pred_indices = logits.argmax(dim=1).detach().cpu().tolist()
            for name, pred_index in zip(batch_names, pred_indices):
                class_name = class_names[pred_index]
                image_names.append(name)
                preds.append(CLASS_NAME_TO_OFFICIAL_LABEL[class_name])

    _write_predictions(image_names, preds)
